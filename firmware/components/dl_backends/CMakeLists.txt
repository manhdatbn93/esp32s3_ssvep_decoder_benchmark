# DL backends for TFLite Micro replay. Artifacts come from firmware/generated.
get_filename_component(FIRMWARE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)

if(CMAKE_BUILD_EARLY_EXPANSION)
    idf_component_register(REQUIRES common)
    return()
endif()

include("${FIRMWARE_ROOT}/cmake/export_artifact_root.cmake")
include("${FIRMWARE_ROOT}/cmake/replay_smoke_enable_dl.cmake")

if(NOT DEFINED TARGET_FOLD_SUBJECT_TAG OR "${TARGET_FOLD_SUBJECT_TAG}" STREQUAL "")
    if(DEFINED SSVEP_SELECTED_TAG)
        set(TARGET_FOLD_SUBJECT_TAG "${SSVEP_SELECTED_TAG}")
    else()
        message(FATAL_ERROR "TARGET_FOLD_SUBJECT_TAG / SSVEP_SELECTED_TAG not set")
    endif()
endif()
set(_TAG "${TARGET_FOLD_SUBJECT_TAG}")

set(DL_SRCS "")
set(DL_PRIV_REQ "")
set(DL_EMBED "")
set(_DL_INCLUDE_DIRS ".")
list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/vectors")

if(SMOKE_ENABLE_DL)
    list(APPEND DL_SRCS "tflite_backend.cpp")
    if(SMOKE_ENABLE_EEGNET_F32 OR SMOKE_ENABLE_EEGNET_INT8)
        list(APPEND DL_SRCS "eegnet_smoke.c")
    endif()
    if(SMOKE_ENABLE_CCNN_F32 OR SMOKE_ENABLE_CCNN_INT8)
        list(APPEND DL_SRCS "ccnn_preprocess.c" "ccnn_smoke.c")
    endif()
    if(SMOKE_ENABLE_CONVCA_F32 OR SMOKE_ENABLE_CONVCA_INT8)
        list(APPEND DL_SRCS "convca_smoke.c")
    endif()
    if(SMOKE_ENABLE_FBTCNN_F32 OR SMOKE_ENABLE_FBTCNN_INT8)
        list(APPEND DL_SRCS "fbtcnn_preprocess.c" "fbtcnn_smoke.c")
    endif()
    list(APPEND DL_PRIV_REQ heap esp_psram)

    if(SMOKE_ENABLE_EEGNET_INT8)
        set(_EEGNET_I8_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_eegnet_${_TAG}_int8/eegnet_int8.tflite")
        list(APPEND DL_EMBED "${_EEGNET_I8_EMBED}")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_eegnet_${_TAG}_int8")
        message(STATUS "[CMake] embedding EEGNet-int8: ${_EEGNET_I8_EMBED}")
    endif()
    if(SMOKE_ENABLE_EEGNET_F32)
        set(_EEGNET_F32_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_eegnet_${_TAG}/eegnet_float32.tflite")
        list(APPEND DL_EMBED "${_EEGNET_F32_EMBED}")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_eegnet_${_TAG}")
        message(STATUS "[CMake] embedding EEGNet-f32: ${_EEGNET_F32_EMBED}")
    endif()
    if(SMOKE_ENABLE_CCNN_F32)
        list(APPEND DL_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_ccnn_${_TAG}/ccnn_float32.tflite")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_ccnn_${_TAG}")
    endif()
    if(SMOKE_ENABLE_CCNN_INT8)
        list(APPEND DL_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_ccnn_${_TAG}_int8/ccnn_int8.tflite")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_ccnn_${_TAG}_int8")
    endif()
    if(SMOKE_ENABLE_CONVCA_F32)
        list(APPEND DL_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_convca_${_TAG}/convca_float32.tflite")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_convca_${_TAG}")
    endif()
    if(SMOKE_ENABLE_CONVCA_INT8)
        list(APPEND DL_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_convca_${_TAG}_int8/convca_int8.tflite")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_convca_${_TAG}_int8")
    endif()
    if(SMOKE_ENABLE_FBTCNN_INT8)
        list(APPEND DL_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_fbtcnn_${_TAG}_int8/fbtcnn_int8.tflite")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_fbtcnn_${_TAG}_int8")
    endif()
    if(SMOKE_ENABLE_FBTCNN_F32)
        list(APPEND DL_EMBED "${EXPORT_ARTIFACT_ROOT}/12jfpm_fbtcnn_${_TAG}/fbtcnn_float32.tflite")
        list(APPEND _DL_INCLUDE_DIRS "${EXPORT_ARTIFACT_ROOT}/12jfpm_fbtcnn_${_TAG}")
    endif()

    set(_DL_ENABLED "")
    if(SMOKE_ENABLE_EEGNET_F32)
        list(APPEND _DL_ENABLED "EEGNet-f32")
    endif()
    if(SMOKE_ENABLE_EEGNET_INT8)
        list(APPEND _DL_ENABLED "EEGNet-int8")
    endif()
    if(SMOKE_ENABLE_CCNN_F32)
        list(APPEND _DL_ENABLED "CCNN-f32")
    endif()
    if(SMOKE_ENABLE_CCNN_INT8)
        list(APPEND _DL_ENABLED "CCNN-int8")
    endif()
    if(SMOKE_ENABLE_CONVCA_F32)
        list(APPEND _DL_ENABLED "ConvCA-f32")
    endif()
    if(SMOKE_ENABLE_FBTCNN_F32)
        list(APPEND _DL_ENABLED "FBtCNN-f32")
    endif()
    if(SMOKE_ENABLE_FBTCNN_INT8)
        list(APPEND _DL_ENABLED "FBtCNN-int8")
    endif()
    string(JOIN ", " _DL_ENABLED_STR ${_DL_ENABLED})
    message(STATUS "dl_backends: tag=${_TAG} root=${EXPORT_ARTIFACT_ROOT} enabled=${_DL_ENABLED_STR}")
    foreach(_emb IN LISTS DL_EMBED)
        if(NOT EXISTS "${_emb}")
            message(FATAL_ERROR "Missing embedded TFLite file: ${_emb}")
        endif()
        message(STATUS "dl_backends: embed ${_emb}")
    endforeach()
else()
    message(STATUS "dl_backends: tag=${_TAG} (stub only)")
endif()

if(NOT DL_SRCS)
    list(APPEND DL_SRCS "dl_stub.c")
endif()

foreach(_inc IN LISTS _DL_INCLUDE_DIRS)
    if(NOT "_${_inc}" STREQUAL "_." AND NOT IS_DIRECTORY "${_inc}")
        message(FATAL_ERROR
            "dl_backends missing include directory: ${_inc}\n"
            "  Re-stage the selected backend into firmware/generated/")
    endif()
endforeach()

idf_component_register(
    SRCS ${DL_SRCS}
    INCLUDE_DIRS ${_DL_INCLUDE_DIRS}
    REQUIRES common
    PRIV_REQUIRES ${DL_PRIV_REQ}
    EMBED_FILES ${DL_EMBED}
)
